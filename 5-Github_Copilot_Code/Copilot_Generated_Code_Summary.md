### Snippet 1: Distributed Lock & Concurrency Manager (`slotLockManager.js`)
* **Prompt to Copilot**: *"Generate an Express.js middleware using Redis to handle pessimistic slot reservation locking with under 200ms lock latency and automatic 30s TTL expiry to prevent double bookings."*

```javascript
// Generated by GitHub Copilot
const redis = require('redis');
const client = redis.createClient();

async function reserveSlotLock(req, res, next) {
  const { slotId, studentId } = req.body;
  const lockKey = `lock:slot:${slotId}`;
  
  // Acquire distributed lock with 30s expiration
  const acquired = await client.set(lockKey, studentId, {
    NX: true, // Only set if key does not exist
    PX: 30000 // Expire after 30,000 ms
  });

  if (!acquired) {
    return res.status(409).json({
      error: "CONCURRENT_BOOKING_COLLISION",
      message: "This slot is currently being reserved by another user. Please choose a different slot."
    });
  }

  req.slotLockKey = lockKey;
  next();
}

module.exports = { reserveSlotLock };
```

### Snippet 2: Domain Recommendation & Matching Engine (`mentorMatcher.js`)
* **Prompt to Copilot**: *"Write a function to rank alumni mentors based on student preferred domain tags (Embedded Systems, Cloud, AI) and mentor ratings."*

```javascript
// Generated by GitHub Copilot
function recommendMentors(mentors, studentDomain, minRating = 4.0) {
  return mentors
    .filter(m => m.domains.includes(studentDomain) && m.averageRating >= minRating)
    .sort((a, b) => b.totalSessionsCompleted - a.totalSessionsCompleted);
}

module.exports = { recommendMentors };
```
